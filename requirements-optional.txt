# =============================================================================
# OPTIONAL heavy dependencies - NOT required to run the app
# =============================================================================
# The backend imports every library below inside a try/except. If they are
# missing, the app boots normally and the affected endpoints return a clear
# "feature unavailable" message instead of crashing:
#
#   * /api/ai/*            -> 503 "facial recognition library not installed"
#   * /api/camera/*        -> wireless camera streaming disabled
#   * voice notifications  -> silently skipped
#
# Install them only on a machine with a compiler and the right system
# libraries. On PythonAnywhere the free/shared tiers do not have them, so
# leave this file alone there.
#
#   Ubuntu/Debian prerequisites:
#       sudo apt-get install -y cmake build-essential libboost-all-dev \
#           libopenblas-dev liblapack-dev libx11-dev libgtk-3-dev espeak1
#
#   pip install -r requirements-optional.txt
# =============================================================================

# Face detection + 128-d encodings. Depends on dlib.
face-recognition==1.3.0
face_recognition_models==0.3.0

# 68-point landmark predictor used for face alignment and blink liveness.
# `pip install dlib` compiles from source and takes 10-20 minutes.
dlib==19.24.6

# OpenCV for frame grabbing, denoising and illumination normalisation.
# Needs libGL and libglib2.0 on Linux; use opencv-python-headless instead if
# you are running without a display.
opencv-python==4.10.0.84

# Audio I/O for voice-based attendance confirmation.
# Needs the espeak-ng system package and libsndfile1.
librosa==0.10.2.post1
pydub==0.25.1
soundfile==0.12.1

# Text-to-speech. Needs espeak-ng.
pyttsx3==2.90
